function(h700_source_target target input)
  if(NOT TARGET "${target}")
    message(FATAL_ERROR "Missing source dependency: ${target}")
  endif()
  get_target_property(real "${target}" ALIASED_TARGET)
  if(NOT real)
    set(real "${target}")
  endif()
  get_target_property(imported "${real}" IMPORTED)
  get_target_property(source "${real}" SOURCE_DIR)
  set(expected "${H700_INPUTS_DIR}/${input}")
  cmake_path(IS_PREFIX expected "${source}" NORMALIZE matches)
  if(imported OR NOT matches)
    message(FATAL_ERROR "${target} must be built from staged ${input}, got ${source}")
  endif()
  set_property(GLOBAL APPEND PROPERTY H700_PROVIDER_REPORT "${target}=${real}:source:${source}\n")
endfunction()

function(h700_package_target target input)
  if(NOT TARGET "${target}")
    message(FATAL_ERROR "Missing package dependency: ${target}")
  endif()
  get_target_property(real "${target}" ALIASED_TARGET)
  if(NOT real)
    set(real "${target}")
  endif()
  get_target_property(imported "${real}" IMPORTED)
  get_target_property(kind "${real}" TYPE)
  get_target_property(location "${real}" IMPORTED_LOCATION)
  if(NOT location)
    get_target_property(location "${real}" IMPORTED_LOCATION_RELEASE)
  endif()
  set(expected "${H700_INPUTS_DIR}/${input}")
  cmake_path(IS_PREFIX expected "${location}" NORMALIZE matches)
  if(NOT imported OR NOT kind STREQUAL "STATIC_LIBRARY" OR NOT matches)
    message(FATAL_ERROR "${target} must use the staged static ${input} package, got ${location}")
  endif()
  set_property(GLOBAL APPEND PROPERTY H700_PROVIDER_REPORT "${target}=${real}:package:${location}\n")
endfunction()
